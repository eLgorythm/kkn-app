Setup Tabel Supabase (SQL Editor)

Setup awal Supabase, salin perintah SQL di bawah ini dan jalankan di menu SQL Editor di Dashboard Supabase: 

CREATE TABLE IF NOT EXISTS kkn_rundown (
  id TEXT PRIMARY KEY,
  week INT DEFAULT 1,
  day TEXT,
  time TEXT,
  category TEXT,
  title TEXT,
  location TEXT,
  pj TEXT,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable Row Level Security (RLS)
ALTER TABLE kkn_rundown ENABLE ROW LEVEL SECURITY;

-- Izinkan akses publik untuk pembacaan & pengubahan
CREATE POLICY "Public Full Access" 
ON kkn_rundown 
FOR ALL 
USING (true) 
WITH CHECK (true);
