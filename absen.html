Setup Database Absensi

Jalankan perintah ini di Supabase Dashboard → SQL Editor. Buat tabel anggota, kehadiran, aktifkan RLS, dan masukkan 13 anggota KKN.

CREATE TABLE IF NOT EXISTS anggota (
  nim TEXT PRIMARY KEY,
  nama TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS kehadiran (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nim TEXT,
  nama TEXT,
  tanggal DATE DEFAULT CURRENT_DATE,
  jam TIMESTAMPTZ DEFAULT NOW(),
  source TEXT DEFAULT 'scan'
);

-- RLS
ALTER TABLE anggota ENABLE ROW LEVEL SECURITY;
ALTER TABLE kehadiran ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public Full Access" ON anggota FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Full Access" ON kehadiran FOR ALL USING (true) WITH CHECK (true);
